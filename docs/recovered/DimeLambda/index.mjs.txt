import AWS from 'aws-sdk';
const dynamoDB = new AWS.DynamoDB.DocumentClient();
const s3 = new AWS.S3();

exports.handler = async (event) => {
    console.log("Received event: ", JSON.stringify(event, null, 2));

    for (const record of event.Records) {
        const bucket = record.s3.object;
        const key = decodeURIComponent(record.s3.object.key.replace(/\+/g, ' '));

        try {
            // Retrieve the content of the MiningContext.js file from S3
            const s3Object = await s3.getObject({ Bucket: bucket, Key: key }).promise();
            const currentContent = s3Object.Body.toString('utf-8');

            // Retrieve the existing content from DynamoDB
            const dynamoDBData = await getDynamoDBItem(key);

            if (dynamoDBData && dynamoDBData.Item) {
                // DynamoDB item exists, compare the content
                if (dynamoDBData.Item.content !== currentContent) {
                    // Values have changed, update DynamoDB
                    const params = {
                        TableName: 'YourDynamoDBTableName',
                        Item: {
                            id: key,
                            content: currentContent,
                        },
                    };

                    await dynamoDB.put(params).promise();
                    console.log(`Data successfully updated in DynamoDB for ${key}`);
                } else {
                    console.log(`No change in values for ${key}`);
                }
            } else {
                // DynamoDB item doesn't exist, create a new one
                const params = {
                    TableName: 'YourDynamoDBTableName',
                    Item: {
                        id: key,
                        content: currentContent,
                    },
                };

                await dynamoDB.put(params).promise();
                console.log(`New DynamoDB item created for ${key}`);
            }
        } catch (error) {
            console.error(`Error processing S3 object ${key}: `, error);
        }
    }

    return {
        statusCode: 200,
        body: JSON.stringify('Data processed and updated/created in DynamoDB'),
    };
};

async function getDynamoDBItem(key) {
    const params = {
        TableName: 'YourDynamoDBTableName',
        Key: {
            id: key,
        },
    };

    return dynamoDB.get(params).promise();
}
